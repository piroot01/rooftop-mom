\section{Reduction to two dimensions}
\label{sec:reduction}

The integrand of a moment depends on the four coordinates only through the two differences $x-x'$ and $y-y'$, and the
weights are polynomials. Integrating out the ``position along the diagonal'' in each direction leaves a two-dimensional
integral with piecewise polynomial weights. The coincident-cell transformation of~\cite{machacek2025method}
(Proposition~2.3 in Section~2.6, regions I--IV), which follows~\cite{Bao2020}, is the special case $\offs=\V{0}$.

\lemmaTitle{One-dimensional correlation}{correlation} Let $\varphi,\psi\in L^\infty(0,1)$, $\delta\in\Rn$, and let
$h:\Rn\to\C$ be locally integrable. Then
\begin{equation}
    \label{eq:correlation}
    \begin{gathered}
    \int_0^1\!\!\int_0^1 \varphi(\xi)\,\psi(\tau)\,h(\xi-\tau-\delta)\D\tau\D\xi
    = \int_{-1}^{1} W_{\varphi\psi}(\nu)\,h(\nu-\delta)\D\nu,\\
    W_{\varphi\psi}(\nu) = \int_{\max(0,\nu)}^{\min(1,1+\nu)} \varphi(\xi)\,\psi(\xi-\nu)\D\xi,
    \end{gathered}
\end{equation}
and both integrals converge absolutely.

\vspace{\baselineskip}
\proof The linear map $(\xi,\tau)\mapsto(\xi,\nu)$, $\nu=\xi-\tau$, has determinant one. It maps the unit square onto the parallelogram (Figure~\ref{fig:correlation})
\begin{equation}
    \{(\xi,\nu):\,-1\le\nu\le1,\ \max(0,\nu)\le\xi\le\min(1,1+\nu)\},
\end{equation}
because $0\le\tau=\xi-\nu\le1$ is equivalent to $\nu\le\xi\le1+\nu$.

By Tonelli's theorem applied to $\ABS{\varphi}\ABS{\psi}\ABS{h}$, the left-hand side is absolutely convergent, since
\begin{equation}
    \int_0^1\!\!\int_0^1 \ABS{\varphi(\xi)\,\psi(\tau)\,h(\xi-\tau-\delta)}\D\tau\D\xi
    \le \NORM{\varphi}_\infty\NORM{\psi}_\infty\int_{-1}^{1}(1-\ABS{\nu})\,\ABS{h(\nu-\delta)}\D\nu < \infty .
\end{equation}
Fubini's theorem then allows the integration in $\xi$ first, for fixed $\nu$, which gives $W_{\varphi\psi}(\nu)$. \qed

\coroTitle{Weights of the rooftop moments}{weights} For $\varphi(\xi)=\xi^\alpha$ and $\psi(\tau)=\tau^\beta$,
$\alpha,\beta\in\{0,1\}$, write $W_{\alpha\beta}=W_{\varphi\psi}$. On $I_-=[-1,0]$ and $I_+=[0,1]$ these are the
polynomials
\begin{equation}
    \label{eq:weights}
    \renewcommand{\arraystretch}{2.2}
    \begin{array}{l|cc}
                 & \nu\in I_-                         & \nu\in I_+ \\ \hline
        W_{00}   & 1+\nu                              & 1-\nu \\
        W_{10}   & \dfrac12(1+\nu)^2                 & \dfrac12(1-\nu^2) \\
        W_{01}   & \dfrac12(1-\nu^2)                 & \dfrac12(1-\nu)^2 \\
        W_{11}   & \dfrac16(1+\nu)^2(2-\nu)          & \dfrac16(2-3\nu+\nu^3)
    \end{array}
\end{equation}
They are continuous on $[-1,1]$, $W_{00}$ and $W_{11}$ are even, and $W_{01}(\nu)=W_{10}(-\nu)$.

\vspace{\baselineskip}
\proof For $\nu\in I_+$ the limits in~\eqref{eq:correlation} are $\nu\le\xi\le1$:
\begin{equation}
    \begin{aligned}
    W_{00} &= \int_\nu^1\!\D\xi = 1-\nu, &
    W_{10} &= \int_\nu^1\!\xi\D\xi = \frac12(1-\nu^2),\\
    W_{01} &= \int_\nu^1\!(\xi-\nu)\D\xi = \frac12(1-\nu)^2, &
    W_{11} &= \int_\nu^1\!\xi(\xi-\nu)\D\xi = \frac16\bigl(2-3\nu+\nu^3\bigr).
    \end{aligned}
\end{equation}
For $\nu\in I_-$ they are $0\le\xi\le1+\nu$:
\begin{equation}
    \begin{aligned}
    W_{00} &= \int_0^{1+\nu}\!\D\xi = 1+\nu, &
    W_{10} &= \int_0^{1+\nu}\!\xi\D\xi = \frac12(1+\nu)^2,\\
    W_{01} &= \int_0^{1+\nu}\!(\xi-\nu)\D\xi = \frac12(1-\nu^2), &
    W_{11} &= \int_0^{1+\nu}\!\xi(\xi-\nu)\D\xi = \frac16(1+\nu)^2(2-\nu).
    \end{aligned}
\end{equation}
At $\nu=0$ both sides give the values $1$, $1/2$, $1/2$ and $1/3$, so the weights are continuous. For the symmetries,
the substitution $s=\xi+\nu$ in~\eqref{eq:correlation} gives $W_{\varphi\psi}(-\nu)=W_{\psi\varphi}(\nu)$. Hence
$W_{01}(\nu)=W_{10}(-\nu)$, and $W_{00}$, $W_{11}$ are even; for $W_{11}$ compare
\begin{equation}
    \frac16(1-\nu)^2(2+\nu)=\frac16\bigl(2-3\nu+\nu^3\bigr).
\end{equation}
\qed

\theoTitle{Two-dimensional representation}{reduction} Let $p>-2$, $\offs=(d_x,d_y)\in\Zn^2$ and
$\alpha,\beta\in\{0,1\}$, and put
\begin{equation}
    \rho_{\offs}(\nu,\eta) = \sqrt{a^2(\nu-d_x)^2+b^2(\eta-d_y)^2}.
\end{equation}
Then
\begin{equation}
    \label{eq:reduction}
    \begin{aligned}
    \Mom{p}{\alpha\beta}{\offs} &= \int_{-1}^{1}\!\!\int_{-1}^{1} W_{\alpha\beta}(\nu)\,W_{00}(\eta)\,
        \rho_{\offs}(\nu,\eta)^p\D\eta\D\nu,\\
    \Nom{p}{\alpha\beta}{\offs} &= \int_{-1}^{1}\!\!\int_{-1}^{1} W_{00}(\nu)\,W_{\alpha\beta}(\eta)\,
        \rho_{\offs}(\nu,\eta)^p\D\eta\D\nu .
    \end{aligned}
\end{equation}
The same holds for a general admissible kernel $K$, with $K(\rho_{\offs})$ in place of $\rho_{\offs}^p$.

\vspace{\baselineskip}
\proof Parametrize $\V{r}=(a\xi_1,b\eta_1)\in\Om{\V{0}}$ and $\V{r}'=\bigl(a(d_x+\xi_2),\,b(d_y+\eta_2)\bigr)\in\Om{\offs}$
with $\xi_1,\eta_1,\xi_2,\eta_2\in[0,1]$. Then $\D A'\D A=(ab)^2\D\eta_2\D\xi_2\D\eta_1\D\xi_1$, which cancels the
normalization of Definition~\defRef{moments}, $\xi(\V{r})=\xi_1$, $\xi(\V{r}')=\xi_2$, and
\begin{equation}
    \NORM{\V{r}-\V{r}'} = \sqrt{a^2(\xi_1-\xi_2-d_x)^2+b^2(\eta_1-\eta_2-d_y)^2}.
\end{equation}
The moment is therefore the integral over the unit cube $[0,1]^4$
\begin{equation}
    \label{eq:fourfold}
    \Mom{p}{\alpha\beta}{\offs} = \int_0^1\!\!\int_0^1\!\!\int_0^1\!\!\int_0^1
    \xi_1^{\alpha}\,\xi_2^{\beta}\,\Bigl(a^2(\xi_1-\xi_2-d_x)^2+b^2(\eta_1-\eta_2-d_y)^2\Bigr)^{p/2}
    \D\eta_2\D\xi_2\D\eta_1\D\xi_1 .
\end{equation}
The integrand depends on the coordinates of the two directions only through the differences $\xi_1-\xi_2$ and
$\eta_1-\eta_2$, and the weight depends on the $x$-coordinates only. The proof integrates out one difference after the
other with Lemma~\lemmaRef{correlation}, first in $x$ and then in $y$.

First, the integral converges absolutely. The function $\V{z}\mapsto\NORM{\V{z}}^p$ is locally integrable on $\Rn^2$ for $p>-2$
(polar coordinates). Lemma~\lemmaRef{correlation}, applied in each direction to the nonnegative kernel by Tonelli's
theorem, gives
\begin{equation}
    \int\!\!\int\!\!\int\!\!\int \ABS{\cdot}
    = \int_{-1}^1\!\!\int_{-1}^1(1-\ABS{\nu})(1-\ABS{\eta})\,\rho_{\offs}^p
    \le \int_{-1}^1\!\!\int_{-1}^1\rho_{\offs}^p < \infty .
\end{equation}
Fubini's theorem therefore allows the four integrations of~\eqref{eq:fourfold} in any order.

Second, the $x$-direction. Fix $\eta_1$ and $\eta_2$, and write $c=b(\eta_1-\eta_2-d_y)$ for the separation of the two
points in $y$. The integrals over $\xi_1$ and $\xi_2$ in~\eqref{eq:fourfold} are then
\begin{equation}
    \int_0^1\!\!\int_0^1 \xi_1^{\alpha}\,\xi_2^{\beta}\,\bigl(a^2(\xi_1-\xi_2-d_x)^2+c^2\bigr)^{p/2}\D\xi_2\D\xi_1 .
\end{equation}
This is the left-hand side of~\eqref{eq:correlation} with $\varphi(\xi)=\xi^\alpha$, $\psi(\tau)=\tau^\beta$,
$\delta=d_x$ and the function of one real variable
\begin{equation}
    h(s)=\bigl(a^2s^2+c^2\bigr)^{p/2}, \qquad s\in\Rn .
\end{equation}
In the lemma $h$ is evaluated at $s=\xi_1-\xi_2-d_x$, so $as$ is the separation of the two points in $x$ and
$a^2s^2+c^2=\NORM{\V{r}-\V{r}'}^2$.
For $c\ne0$ the function $h$ is continuous, hence locally integrable. The pairs with $c=0$, that is
$\eta_1-\eta_2=d_y$, form a line in the $(\eta_1,\eta_2)$-square and do not contribute to the outer integral. The
lemma, with the weight $W_{\varphi\psi}=W_{\alpha\beta}$ of Corollary~\coroRef{weights}, gives
\begin{equation}
    \int_0^1\!\!\int_0^1 \xi_1^{\alpha}\,\xi_2^{\beta}\,\bigl(a^2(\xi_1-\xi_2-d_x)^2+c^2\bigr)^{p/2}\D\xi_2\D\xi_1
    = \int_{-1}^{1} W_{\alpha\beta}(\nu)\,\bigl(a^2(\nu-d_x)^2+c^2\bigr)^{p/2}\D\nu .
\end{equation}
The two $x$-coordinates are replaced by their difference $\nu$, and the weights $\xi_1^\alpha$, $\xi_2^\beta$ by the
single weight $W_{\alpha\beta}(\nu)$. With $c=b(\eta_1-\eta_2-d_y)$ inserted back,
\begin{equation}
    \Mom{p}{\alpha\beta}{\offs} = \int_{-1}^{1} W_{\alpha\beta}(\nu)\Biggl(\int_0^1\!\!\int_0^1
    \Bigl(a^2(\nu-d_x)^2+b^2(\eta_1-\eta_2-d_y)^2\Bigr)^{p/2}\D\eta_2\D\eta_1\Biggr)\D\nu ,
\end{equation}
where the $\nu$-integration has been moved outside by Fubini's theorem.

Third, the $y$-direction. Fix $\nu\ne d_x$; the single value $\nu=d_x$ again does not contribute. The inner integral
in the last display is the left-hand side of~\eqref{eq:correlation} once more, now in the variables $(\eta_1,\eta_2)$
with $\varphi=\psi=1$, $\delta=d_y$ and
\begin{equation}
    h(s)=\bigl(a^2(\nu-d_x)^2+b^2s^2\bigr)^{p/2}, \qquad s\in\Rn ,
\end{equation}
where $s$ now stands for $\eta_1-\eta_2-d_y$, so $bs$ is the separation of the two points in $y$.
For the weights $1$ and $1$ the weight of the lemma is $W_{00}$, so
\begin{equation}
    \int_0^1\!\!\int_0^1\Bigl(a^2(\nu-d_x)^2+b^2(\eta_1-\eta_2-d_y)^2\Bigr)^{p/2}\D\eta_2\D\eta_1
    = \int_{-1}^{1} W_{00}(\eta)\,\rho_{\offs}(\nu,\eta)^p\D\eta .
\end{equation}
Inserting this into the previous display gives the first line of~\eqref{eq:reduction}.

Finally, the $N$-moments carry the weights $\eta_1^\alpha\eta_2^\beta$ in the $y$-coordinates instead. The same three
steps then give $W_{00}(\nu)$ in $x$ and $W_{\alpha\beta}(\eta)$ in $y$, which is the second line
of~\eqref{eq:reduction}. Nothing in the argument uses the form of the power $\rho^p$ beyond its local integrability, so
a general admissible kernel $K$ gives the same result with $K(\rho_{\offs})$ in place of $\rho_{\offs}^p$. \qed

\vspace{\baselineskip}
\noindent\textbf{Remark.} For $\offs\in\mathcal{T}$ the singular point of $\rho_{\offs}^p$, $p<0$, is
$(\nu,\eta)=(d_x,d_y)$. It is the centre, a midpoint of an edge or a corner of $[-1,1]^2$ (Figure~\ref{fig:regions}). In every case it lies on
the lines $\nu\in\{-1,0,1\}$ and $\eta\in\{-1,0,1\}$, where the weights change their polynomial form. This is the
structural fact behind the closed forms of Section~\ref{sec:touching}.
